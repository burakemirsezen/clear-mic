cmake_minimum_required(VERSION 3.20)
project(ClearMicVoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(MODULE aec3 agc2)
  set(SOURCE "${ROOT}/vendor/webrtc-${MODULE}/src")
  file(GLOB_RECURSE SOURCES "${SOURCE}/*.cc" "${SOURCE}/*.c" "${SOURCE}/*.cpp")
  list(FILTER SOURCES EXCLUDE REGEX ".*_neon\\.cc$")
  add_library(cm_${MODULE} SHARED ${SOURCES} "${ROOT}/native/${MODULE}_bridge.cpp")
  target_compile_options(cm_${MODULE} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>" /FI"${ROOT}/native/webrtc_compat.h" /arch:AVX2 /O2)
  target_compile_definitions(cm_${MODULE} PRIVATE WEBRTC_WIN NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
  target_include_directories(cm_${MODULE} PRIVATE "${SOURCE}" "${SOURCE}/api" "${SOURCE}/base/rtc_base" "${SOURCE}/base" "${SOURCE}/base/system_wrappers/include" "${SOURCE}/audio_processing" "${SOURCE}/absl" "${SOURCE}/base/jsoncpp/include" "${SOURCE}/audio_processing/aec3" "${SOURCE}/audio_processing/resampler" "${SOURCE}/audio_processing/logging" "${SOURCE}/audio_processing/include" "${SOURCE}/audio_processing/utility" "${SOURCE}/rtc_base/experiments" "${SOURCE}/common_audio" "${SOURCE}/common_audio/third_party/ooura/fft_size_128")
  target_link_libraries(cm_${MODULE} PRIVATE winmm ws2_32 ole32)
endforeach()
add_library(cm_reference SHARED "${ROOT}/native/speaker_reference.cpp")
target_compile_definitions(cm_reference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(cm_reference PRIVATE ole32 uuid avrt)
